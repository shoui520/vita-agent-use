cmake_minimum_required(VERSION 3.16)
project(vita_agent_use VERSION 0.1.0 LANGUAGES C)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_library(vau_core STATIC src/module_variable.c src/content_media_paths.c src/sfo_titles.c src/content_delete.c src/content_scope_json.c src/content_sdk.c src/content_media.c src/content_media_catalog.c src/content_media_session.c src/log_watch.c src/log_marker.c src/event_ring.c src/touch_sample.c src/file_ops.c src/input_owner.c src/timeline.c src/stop_monitor.c src/policy.c src/acl_config.c src/tai_config_guard.c src/write_ops.c src/write_changes.c src/upload_wire.c src/native_ops.c src/system_metadata.c src/performance.c src/json.c src/sha256.c src/protocol.c src/input_sequence.c src/http.c src/auth.c src/service.c src/notification.c src/pairing_prompt.c src/pairing_request.c src/connection.c src/ui_job.c)
target_include_directories(vau_core PUBLIC include PRIVATE src)
target_compile_options(vau_core PRIVATE -Wall -Wextra -Werror "$<$<CONFIG:Release>:-O2>")

set(VAU_MBEDTLS_SOURCE "" CACHE PATH "Mbed TLS 3.6.7 source tree for the user transport")
if(VAU_MBEDTLS_SOURCE)
    set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
    set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    set(MBEDTLS_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/src/tls_config.h" CACHE FILEPATH "" FORCE)
    add_subdirectory("${VAU_MBEDTLS_SOURCE}" "transport-dependency" EXCLUDE_FROM_ALL)
    foreach(tls_library mbedcrypto mbedx509 mbedtls)
        target_compile_options(${tls_library} PRIVATE -Os)
    endforeach()
    add_library(vau_transport STATIC src/grant_reply.c src/command_worker.c src/tls_driver.c src/tls_server.c src/pairing_tls.c src/pairing_flow.c)
    target_include_directories(vau_transport PUBLIC src include)
    target_compile_options(vau_transport PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(vau_transport PUBLIC vau_core mbedtls)
endif()

if(CMAKE_CROSSCOMPILING)
    target_compile_definitions(vau_core PUBLIC VAU_NATIVE_FORMAT=1)
    include("${VITASDK}/share/vita.cmake" REQUIRED)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS native365.yml)
    execute_process(COMMAND "${VITASDK}/bin/vita-libs-gen" -c
        "${CMAKE_CURRENT_SOURCE_DIR}/native365.yml" "${CMAKE_CURRENT_BINARY_DIR}/native365-stubs"
        RESULT_VARIABLE native_stubs_result)
    if(NOT native_stubs_result EQUAL 0)
        message(FATAL_ERROR "Native import stub generation failed")
    endif()
    add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/native365-stubs"
        "${CMAKE_CURRENT_BINARY_DIR}/native365-stub-build" EXCLUDE_FROM_ALL)
    add_library(vau_ui_native STATIC src/ui_vita.c src/dialog_callbacks.c src/dialog_vita.c src/pairing_ui.c src/notification_vita.c src/notification_worker.c)
    target_include_directories(vau_ui_native PUBLIC src include)
    target_compile_options(vau_ui_native PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(vau_ui_native PUBLIC vau_core ScePaf_365_stub SceCommonGuiDialog_365_stub SceNotificationUtil_365_stub SceShellSvc_365_stub SceLibKernel_stub SceKernelThreadMgr_stub)
    if(TARGET vau_transport)
        target_sources(vau_transport PRIVATE src/net_vita.c)
        target_link_libraries(vau_transport PUBLIC vau_ui_native SceNet_stub SceKernelThreadMgr_stub)
    endif()
    add_library(vau_native STATIC src/package_install.c src/content_scope.c src/content_media_capture_vita.c src/content_capture.c src/content_media_vita.c src/content_sdk_vita.c src/sfo_titles_vita.c src/content_export.c src/content_vita.c src/content_runtime.c src/content_delete_vita.c src/content_journal.c src/content_inventory.c src/dialog_events.c src/livearea_blob.c src/livearea_layout.c src/sqlite_json.c src/log_watch_vita.c src/livearea.c src/events_vita.c src/files_vita.c src/system_vita.c src/metadata_vita.c src/plugins_vita.c src/performance_vita.c src/touch_vita.c src/pairing_activation.c src/pairing_worker.c src/native_vita.c src/input_vita.c src/service_vita.c src/shell_jpeg.c src/frames_shell.c src/app_registry.c src/sqlite_memory.c src/sqlite_vfs.c src/write_journal.c src/audit_export.c src/audit_vita.c src/staged_upload.c src/upload_commit.c src/config_native.c src/file_tree.c src/file_mutations.c src/trash_purge.c src/write_observe.c src/writes_vita.c src/acl_vita.c src/acl_approval.c)
    target_include_directories(vau_native PUBLIC include PRIVATE src)
    target_compile_options(vau_native PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(vau_native PUBLIC vau_core vau_ui_native SceRegMgr_365_stub k SceVshBridge_stub SceKernelModulemgr_stub SceAppMgr_stub ScePower_stub
        SceIofilemgr_stub SceLibKernel_stub SceKernelThreadMgr_stub taihen_stub SceSysmem_stub SceTouch_stub SceNet_stub SceNetCtl_stub SceWlanBt_stub SceBt_stub SceSysmodule_stub SceRegistryMgr_stub SceAudioIn_stub SceSqlite_365_stub z gcc)
    if(TARGET vau_transport)
        target_link_libraries(vau_native PUBLIC mbedcrypto)
        target_link_libraries(mbedcrypto PUBLIC k gcc)
        target_link_libraries(vau_core PUBLIC k gcc)
        add_executable(vita_agent_shell src/shell_runtime.c src/diagnostics_vita.c)
        target_include_directories(vita_agent_shell PRIVATE src include)
        target_compile_options(vita_agent_shell PRIVATE -Wall -Wextra -Werror -mcpu=cortex-a9)
        target_link_options(vita_agent_shell PRIVATE -nostdlib -Wl,-q)
        target_link_libraries(vita_agent_shell PRIVATE vau_native vau_transport SceProcessmgr_stub)
        vita_create_self(vita_agent_shell.suprx vita_agent_shell CONFIG shell.yml UNSAFE)
        add_executable(vita_agent_loader src/shell_loader.c)
        target_compile_options(vita_agent_loader PRIVATE -Wall -Wextra -Werror -mcpu=cortex-a9)
        target_link_options(vita_agent_loader PRIVATE -nostdlib -Wl,-q)
        target_link_libraries(vita_agent_loader PRIVATE k SceSysmodule_stub SceKernelModulemgr_stub
            SceLibKernel_stub SceKernelThreadMgr_stub SceIofilemgr_stub gcc)
        vita_create_self(vita_agent_loader.suprx vita_agent_loader CONFIG loader.yml UNSAFE)
    endif()
    target_compile_options(vau_core PRIVATE -mcpu=cortex-a9 -mgeneral-regs-only -mno-unaligned-access)
    add_executable(vita_agent_kernel src/kernel.c src/events_kernel.c src/capture_kernel.c src/modules_kernel.c src/performance_kernel.c)
    add_library(vau_touch_kernel STATIC src/touch_kernel.c)
    target_include_directories(vau_touch_kernel PRIVATE include src)
    target_compile_options(vau_touch_kernel PRIVATE -Wall -Wextra -Werror
        -mcpu=cortex-a9 -mgeneral-regs-only -mno-unaligned-access)
    target_link_libraries(vau_touch_kernel PUBLIC vau_core taihenForKernel_stub
        SceModulemgrForKernel_363_stub SceThreadmgrForDriver_stub SceCpuForDriver_stub
        SceSysclibForDriver_stub SceTouchForDriver_stub)
    target_include_directories(vita_agent_kernel PRIVATE include src)
    target_compile_options(vita_agent_kernel PRIVATE -Wall -Wextra -Werror
        -mcpu=cortex-a9 -mgeneral-regs-only -mno-unaligned-access "$<$<CONFIG:Release>:-O2>")
    target_link_options(vita_agent_kernel PRIVATE -nostdlib -Wl,-q)
    target_link_libraries(vita_agent_kernel PRIVATE vau_core vau_touch_kernel
        SceSysclibForDriver_stub SceThreadmgrForDriver_stub SceSysmemForDriver_stub
        SceSysrootForDriver_stub SceCtrlForDriver_stub SceCtrlForDriver_365_stub SceDisplayForDriver_stub SceModulemgrForKernel_363_stub taihenForKernel_stub SceSysmemForKernel_365_stub SceSysrootForKernel_365_stub gcc)
    vita_create_self(vita_agent_kernel.skprx vita_agent_kernel
        CONFIG kernel.yml UNSAFE)
    vita_create_stubs(stubs vita_agent_kernel ${CMAKE_CURRENT_SOURCE_DIR}/kernel.yml KERNEL)
    add_custom_target(vau_stubs ALL DEPENDS libvita_agent_kernel_stub.a
        libvita_agent_kernel_stub_weak.a)
    add_dependencies(vau_native vau_stubs)
    target_link_libraries(vau_native PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/stubs/libvita_agent_kernel_stub.a")
else()
    option(VAU_TESTS "Build local validation tests when available" OFF)
    if(VAU_TESTS)
        enable_testing()
        add_subdirectory(agent/tests)
    endif()
endif()
